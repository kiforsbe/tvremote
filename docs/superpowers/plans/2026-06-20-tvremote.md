# TV Remote Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Samsung TV remote HTML app that runs as a compact Electron window on Windows 11 (with background global shortcuts) and installs as a PWA on iPhone via a local HTTP server — with whitebox unit tests, blackbox HTTP integration tests, and Playwright GUI tests.

**Architecture:** A single `app/remote.js` holds all testable logic and is loaded by `app/index.html` via `<script src="remote.js">`. Electron's `main.js` loads the UI, registers media key global shortcuts, and serves `app/` over HTTP on port 3000. Jest covers unit + blackbox tests; Playwright covers GUI tests against both desktop and mobile (iPhone) viewports.

**Tech Stack:** Electron ^31, Node.js built-in `http`/`fs`/`path`, jimp ^0.22 (icon generation), Jest ^29, @playwright/test ^1.44, vanilla HTML/CSS/JS — no bundler.

## Global Constraints

- Window: 260×680px, not resizable
- HTTP server: port 3000, bound to 0.0.0.0
- Global shortcuts: media keys only (MediaPlayPause, VolumeUp, VolumeDown, VolumeMute)
- No service worker, no bundler, no build step
- No branded app shortcut buttons (Netflix, Disney+, etc.)
- Button touch targets: minimum 44px (iOS HIG) — enforced by Playwright test
- Safe area insets applied to body padding (iPhone notch/home bar)
- Color palette: `--bg #0f0f0f`, `--remote-body #141414`, `--btn #1e1e1e`, `--btn-hover #2a2a2a`, `--text #e0e0e0`, `--text-dim #555`, `--power-red #c0392b`
- Remote body: 180px wide, border-radius 44px
- Nav ring: 116px diameter, dominant focal element
- `samsung_remote.html` (old file) is deleted in Task 4

```js
const Jimp = require('jimp');
const path = require('path');
const fs   = require('fs');

const OUT = path.join(__dirname, '..', 'app', 'icons');
fs.mkdirSync(OUT, { recursive: true });

async function generate(size) {
  const img = new Jimp(size, size, 0x141414ff);

  const cx = size / 2;
  const cy = size / 2;
  const bw = size * 0.55;
  const bh = size * 0.80;
  const br = size * 0.18;

  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const ax = Math.abs(x - cx);
      const ay = Math.abs(y - cy);
      const dx = Math.max(0, ax - (bw / 2 - br));
      const dy = Math.max(0, ay - (bh / 2 - br));
      if (ax <= bw / 2 && ay <= bh / 2 && Math.hypot(dx, dy) <= br) {
        img.setPixelColor(0x2a2a2aff, x, y);
      }
    }
  }

  const ringR = size * 0.15;
  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      if (Math.hypot(x - cx, y - cy) <= ringR) {
        img.setPixelColor(0x1e1e1eff, x, y);
      }
    }
  }

  await img.writeAsync(path.join(OUT, `icon-${size}.png`));
  console.log(`  icon-${size}.png`);
}

(async () => {
  console.log('Generating icons…');
  await generate(192);
  await generate(512);
  console.log('Done.');
})().catch(e => { console.error(e); process.exit(1); });
```

```js
const KEY_MAP = {
  ArrowUp:    'KEY_UP',
  ArrowDown:  'KEY_DOWN',
  ArrowLeft:  'KEY_LEFT',
  ArrowRight: 'KEY_RIGHT',
  Enter:      'KEY_ENTER',
  ' ':        'KEY_ENTER',
  h:          'KEY_HOME',
  b:          'KEY_BACK',
  m:          'KEY_MENU',
  '+':        'KEY_VOLUP',
  '-':        'KEY_VOLDOWN',
  '0':        'KEY_MUTE',
  PageUp:     'KEY_CHUP',
  PageDown:   'KEY_CHDOWN',
  p:          'KEY_POWER',
};

function buildKeyCommand(key) {
  return JSON.stringify({
    method: 'ms.remote.control',
    params: {
      Cmd: 'Click',
      DataOfCmd: key,
      Option: 'false',
      TypeOfRemote: 'SendRemoteKey'
    }
  });
}

let ws = null;

function setStatus(text, state) {
  const el = document.getElementById('status');
  el.textContent = text;
  el.className = state || '';
}

function connect() {
  if (ws) { ws.close(); ws = null; }
  const ip = document.getElementById('tvip').value.trim();
  if (!ip) return;
  setStatus('Connecting…');
  const url =
    `wss://${ip}:8002/api/v2/channels/samsung.remote.control?name=${btoa('TVRemote')}`;
  ws = new WebSocket(url);
  ws.onopen  = () => setStatus('Connected', 'ok');
  ws.onerror = () => setStatus('Error — see cert hint below', 'err');
  ws.onclose = () => { setStatus('Disconnected', 'err'); ws = null; };
}

function sendKey(key) {
  if (!ws || ws.readyState !== WebSocket.OPEN) {
    setStatus('Not connected', 'err');
    return;
  }
  ws.send(buildKeyCommand(key));
}

function openCertPage() {
  const ip = document.getElementById('tvip').value.trim() || '192.168.50.122';
  window.open(`https://${ip}:8002`, '_blank');
}

document.addEventListener('keydown', e => {
  const key = KEY_MAP[e.key];
  if (key) { e.preventDefault(); sendKey(key); }
});

// Exported for Jest — browser ignores this (module is undefined)
if (typeof module !== 'undefined' && module.exports) {
  module.exports = { KEY_MAP, buildKeyCommand };
}
```

// tests/blackbox/server.test.js
/** @jest-environment node */

const http = require('http');

function get(path) {
  return new Promise((resolve, reject) => {
    const req = http.get(`http://localhost:3000${path}`, res => {
      const chunks = [];
      res.on('data', c => chunks.push(c));
      res.on('end', () => resolve({
        status: res.statusCode,
        headers: res.headers,
        body: Buffer.concat(chunks),
      }));
    });
    req.on('error', reject);
    req.setTimeout(5000, () => { req.destroy(new Error('timeout')); });
  });
}

describe('HTTP server', () => {
  test('serves index.html at / with correct content-type', async () => {
    const r = await get('/');
    expect(r.status).toBe(200);
    expect(r.headers['content-type']).toContain('text/html');
    expect(r.body.toString()).toContain('<!DOCTYPE html>');
  });

  test('serves remote.js at /remote.js', async () => {
    const r = await get('/remote.js');
    expect(r.status).toBe(200);
    expect(r.headers['content-type']).toContain('application/javascript');
    expect(r.body.toString()).toContain('buildKeyCommand');

  test('serves manifest.json with correct content-type', async () => {
    const r = await get('/manifest.json');

    expect(r.headers['content-type']).toContain('application/json');

  test('serves icon-192.png as image/png', async () => {

    expect(r.headers['content-type']).toBe('image/png');
    // PNG magic bytes
    expect(r.body[0]).toBe(0x89);

- [ ] **Step 3: Create main.js**

```js
const { app, BrowserWindow, globalShortcut } = require('electron');
const http = require('http');
const fs   = require('fs');
const path = require('path');

const APP_DIR   = path.join(__dirname, 'app');
const HTTP_PORT = 3000;
const WIN_W     = 260;
const WIN_H     = 680;

const MIME = {
  '.html': 'text/html; charset=utf-8',
  '.json': 'application/json',
  '.png':  'image/png',
  '.js':   'application/javascript',
};

// Allow Samsung TV's self-signed WSS certificate
app.on('certificate-error', (event, _wc, _url, _err, _cert, cb) => {
  event.preventDefault();
  cb(true);
});

function startHttpServer() {
  http.createServer((req, res) => {
    const rel  = req.url === '/' ? '/index.html' : req.url;
    const file = path.join(APP_DIR, path.normalize(rel));

    // Prevent directory traversal
    if (!file.startsWith(APP_DIR + path.sep) && file !== APP_DIR) {
      res.writeHead(403);
      return res.end();
    }

    fs.readFile(file, (err, data) => {
      if (err) { res.writeHead(404); return res.end('Not found'); }
      res.writeHead(200, {
        'Content-Type': MIME[path.extname(file)] || 'application/octet-stream',
      });
      res.end(data);
    });
  }).listen(HTTP_PORT, '0.0.0.0', () => {
    console.log(`Remote available at http://<this-pc-ip>:${HTTP_PORT}`);
  });
}

function registerShortcuts(win) {
  const map = {
    MediaPlayPause: 'KEY_PLAY',
    VolumeUp:       'KEY_VOLUP',
    VolumeDown:     'KEY_VOLDOWN',
    VolumeMute:     'KEY_MUTE',
  };

  for (const [accel, key] of Object.entries(map)) {
    globalShortcut.register(accel, () => {
      win.webContents.executeJavaScript(`sendKey('${key}')`).catch(() => {});
    });
  }
}

app.whenReady().then(() => {
  const win = new BrowserWindow({
    width:     WIN_W,
    height:    WIN_H,
    resizable: false,
    webPreferences: {
      nodeIntegration:  false,
      contextIsolation: true,
    },
  });

  win.loadFile(path.join(APP_DIR, 'index.html'));
  startHttpServer();
  registerShortcuts(win);
});

app.on('window-all-closed', () => {
  globalShortcut.unregisterAll();
  app.quit();
});
```

**Certificate workaround (first time only):**
1. In the app, tap the **"accept TV certificate"** button in the cert hint
